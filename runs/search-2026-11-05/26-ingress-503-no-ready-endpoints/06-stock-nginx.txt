# Each location produces one candidate cause on stock nginx.
log_format lab '$remote_addr "$request" $status rt=$request_time us=$upstream_status ua=$upstream_addr lrs=$limit_req_status';
limit_req_zone $binary_remote_addr zone=one:1m rate=2r/s;
limit_req_zone $binary_remote_addr zone=dry:1m rate=2r/s;
limit_req_zone $binary_remote_addr zone=four29:1m rate=2r/s;
limit_conn_zone $binary_remote_addr zone=conn:1m;
upstream alldown { server backend:8080 down; }
upstream closed { server 127.0.0.1:9; }
upstream onlymaxconns { server backend:8080 max_conns=1; }
upstream pair { server faila:8080; server failb:8080; }
server {
    listen 80;
    access_log /dev/stdout lab;
    location /ok { proxy_pass http://backend:8080/ok; }
    location /refused { proxy_pass http://closed/; }
    location /alldown { proxy_pass http://alldown/; }
    location /maxconns { proxy_pass http://onlymaxconns/sleep/2; }
    location /limit { limit_req zone=one; proxy_pass http://backend:8080/ok; }
    location /dry { limit_req zone=dry; limit_req_dry_run on; proxy_pass http://backend:8080/ok; }
    location /four29 { limit_req zone=four29; limit_req_status 429; proxy_pass http://backend:8080/ok; }
    location /conn { limit_conn conn 1; proxy_pass http://backend:8080/sleep/2; }
    location /maint { return 503; }
    location /app503 { proxy_pass http://backend:8080/fail503; }
    location /app503-server { proxy_pass http://backend:8080/fail503; proxy_pass_header Server; }
    location /intercept {
        proxy_pass http://backend:8080/fail503;
        proxy_intercept_errors on;
        error_page 503 /sorry;
    }
    location = /sorry { internal; return 503 "we will be back soon\n"; }
    location /pair { proxy_pass http://pair/; proxy_next_upstream error timeout http_503; }
}
